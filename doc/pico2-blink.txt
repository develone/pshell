    //int led_pin = PICO_DEFAULT_LED_PIN;
    int led_pin = 25;
    gpio_init(led_pin);
    gpio_set_dir(led_pin, GPIO_OUT);
    int tic = 0;
    int i;    
    for (i=0;i<25;i++) {
        printf("i = %d led = %d \n",i,tic);
        gpio_put(led_pin, tic ^= 1);
        if (getchar_timeout_us(500000) == 3)
            break;
    }
    gpio_put(led_pin, 0);
    return 0;
}
